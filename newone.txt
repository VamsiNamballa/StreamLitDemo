DROP TABLE IF EXISTS bookings;
DROP TABLE IF EXISTS shows;
DROP TABLE IF EXISTS movies;
DROP TABLE IF EXISTS customer;

CREATE TABLE customer (
    customer_id INT PRIMARY KEY,
    customer_name VARCHAR(100),
    city VARCHAR(50),
    age INT
);

CREATE TABLE movies (
    movie_id INT PRIMARY KEY,
    movie_name VARCHAR(100),
    genre VARCHAR(50),
    language VARCHAR(50),
    duration_minutes INT
);

CREATE TABLE shows (
    show_id INT PRIMARY KEY,
    movie_id INT,
    theatre_name VARCHAR(100),
    show_date DATE,
    ticket_price NUMERIC(10,2),
    FOREIGN KEY (movie_id) REFERENCES movies(movie_id)
);

CREATE TABLE bookings (
    booking_id INT PRIMARY KEY,
    customer_id INT,
    show_id INT,
    tickets_booked INT,
    booking_date DATE,
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id),
    FOREIGN KEY (show_id) REFERENCES shows(show_id)
);

SELECT b.booking_id, c.customer_name, b.show_id, b.tickets_booked
FROM bookings b
left join customer c
on c.customer_id=b.customer_id;

-- Q12. Show show details with movie name. --
SELECT s.show_id, m.movie_name, s.theatre_name, s.show_date, s.ticket_price
FROM shows s
JOIN movies m
on s.movie_id=m.movie_id;

-- Q13. Show full booking details. --
-- booking_id, customer_name, movie_name, theatre_name, tickets_booked, ticket_price
SELECT b.booking_id, c.customer_name, m.movie_name, s.theatre_name, b.tickets_booked, s.ticket_price
FROM movies m 
JOIN shows s
on m.movie_id=s.movie_id
JOIN bookings b 
on b.show_id=s.show_id
JOIN customer c
on c.customer_id=b.customer_id;

-- Q14. Show total amount for each booking. --
-- booking_id, customer_name, movie_name, tickets_booked, ticket_price, total_amount --
-- Formula= tickets_booked * ticket_price --
SELECT b.booking_id, c.customer_name, m.movie_name, b.tickets_booked, s.ticket_price,b.booking_id, s.show_id, b.tickets_booked * s.ticket_price AS Total_amount
FROM bookings b
join shows s
on b.show_id=s.show_id
JOIN movies m 
on m.movie_id=s.movie_id
JOIN customer c
on c.customer_id=b.customer_id;

-- Q16. Find total tickets booked per movie. --
-- movie_name, total_tickets --
SELECT m.movie_name, SUM(b.tickets_booked) as total_tickets
FROM movies m
join shows s
on s.movie_id=m.movie_id
join bookings b
on s.show_id=b.show_id
GROUP BY m.movie_name;

-- Q17. Find total revenue per movie. --
-- SUM(tickets_booked * ticket_price) --
SELECT m.movie_name, SUM(b.tickets_booked * s.ticket_price) as total_revenue
FROM movies m
join shows s
on s.movie_id=m.movie_id
join bookings b
on b.show_id=s.show_id
GROUP BY m.movie_name;

-- Q23. Find cities with more than 2 bookings. --
SELECT c.city,
COUNT(b.booking_id) as total_bookings
FROM shows s
JOIN bookings b
on s.show_id=b.show_id
join customer c
on c.customer_id=b.customer_id
group by c.city
having COUNT(b.booking_id)>2;

-- Q24. Using a CTE, first calculate total revenue per movie, 
-- then show only movies with revenue greater than 1000.

WITH movie_revenue as(
	SELECT m.movie_name, SUM(b.tickets_booked * s.ticket_price) as total_revenue
FROM movies m
join shows s
on s.movie_id=m.movie_id
join bookings b
on b.show_id=s.show_id
GROUP BY m.movie_name
)
SELECT *
FROM movie_revenue mr
where mr.total_revenue>1000;

-- Q25. Using a CTE, calculate total tickets booked per customer, 
-- then show customers with total tickets greater than 3.

WITH tickets_per_customer as(
SELECT c.customer_name, COUNT(b.booking_id) as counter
FROM bookings b
join customer c
on b.customer_id=c.customer_id
group by c.customer_name
)
SELECT * 
from tickets_per_customer tpc
where tpc.counter>3;

-- Q26. Using a CTE, calculate total amount for every booking, 
-- then filter bookings where total amount is greater than 700.

WITH CTE AS(
SELECT b.booking_id, b.tickets_booked * s.ticket_price as total_amount
FROM bookings b
join shows s
on b.show_id=s.show_id
)
SELECT * 
FROM CTE 
WHERE CTE.total_amount>700;

-- Q27. Rank movies by total revenue. --

SELECT m.movie_name, 
SUM(s.ticket_price*b.tickets_booked) as total_revenue,
RANK() OVER(ORDER BY SUM(s.ticket_price*b.tickets_booked) DESC)
FROM movies m
join shows s
on m.movie_id=s.movie_id
join bookings b
on b.show_id=s.show_id
GROUP BY m.movie_name;

-- Q28. Find the highest revenue movie. --
SELECT m.movie_name, 
SUM(s.ticket_price*b.tickets_booked) as total_revenue,
RANK() OVER(ORDER BY SUM(s.ticket_price*b.tickets_booked) DESC) as rev_rank
FROM movies m
join shows s
on m.movie_id=s.movie_id
join bookings b
on b.show_id=s.show_id
GROUP BY m.movie_name
order by rev_rank;

-- Q28. Find the highest revenue movie.

FROM movies m

-- Q29. Rank customers by total tickets booked. --

WITH custtickets as(
SELECT c.customer_name as cname, SUM(b.tickets_booked) booked_tickets
FROM customer c
join bookings b
on c.customer_id=b.customer_id
group by c.customer_name
)
SELECT ctkt.cname, ctkt.booked_tickets,
RANK() OVER (ORDER BY ctkt.booked_tickets)
FROM custtickets ctkt;
